genetic_code=("TACGGCTTACTAGCTGGCAAATCCTGAATCCGGACT")

total_bases=len(genetic_code)
#dictionary
genome_calculator={}
#loop
for base in genetic_code:
    if base in genome_calculator:
        genome_calculator[base]=genome_calculator[base]+1
    else:
        genome_calculator[base]=1

#final dashboard

print("Genetic code analysis complete")
for genome, count in genome_calculator.items():
    print(f"total of genetic letter: {genome}: {count}")
    percentage=round((count/total_bases)*100,2)
    print(f"percentage of genetic code:{genome}: ({percentage})")
